━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Vaybelix; çok ajanlı (multi-agent) yapay zeka destekli, tam otomatik haber ve
rehber içeriği üretip yayınlayan bir içerik platformudur. Süpervizör, editör ve
ziyaretçi için farklı arayüzler sunar. Finansal takip, reklam yönetimi, SEO
optimizasyonu ve web push bildirim gibi özellikler içerir.

  5. OTOMASYON SİSTEMİ — PIPELINE MİMARİSİ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  İki ayrı pipeline mevcuttur:

  A) HABER PİPELINE (news)
  ─────────────────────────
  Adım  1 → ScoutAgent      : RSS/NewsAPI/Reddit/YouTube kaynaklarını tara,
                               haber başlıklarını listele
  Adım  2 → WriterAgent     : Seçilen haberi taslak makaleye dönüştür
  Adım  3 → EditorAgent     : Dil, stil ve tutarlılık düzenlemesi yap
  Adım  4 → SeoAgent        : SEO başlığı, meta açıklaması, anahtar kelime
                               optimizasyonu
  Adım  5 → MediaAgent      : Kapak görseli üret (Sharp/Cloudflare Worker),
                               Pexels'ten makale içi görseller getir
  Adım  6 → PublisherAgent  : Veritabanına kaydet, görsel enjekte et, yayınla

  B) REHBER PİPELINE (guide)
  ──────────────────────────
  Adım  1 → GuideScoutAgent    : Rehber konusu ve anahtar kelimeler belirle
  Adım  2 → GuideWriterAgent   : Uzun biçimli rehber içeriği yaz
  Adım  3 → GuideEditorAgent   : İçeriği düzenle, yapıyı iyileştir
  Adım  4 → GuideSeoAgent      : Rehber için SEO optimizasyonu
  Adım  5 → MediaAgent         : Görsel üretimi (paylaşımlı)
  Adım  6 → GuidePublisherAgent: Affiliate linkleri yerleştir, ToC/JSON-LD
                                  ekle, görselleri enjekte et, yayınla

  ─────────────────────────────

  newsapi.ts    NewsAPI.org API entegrasyonu

  7. GÜVENLİK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ŞİFRELEME
  ──────────
  • Algoritma: AES-256-CBC (Node.js crypto modülü)
  • Anahtar: 256-bit (64 hex karakter) ENCRYPTION_KEY ortam değişkeni
  • Format: iv_hex:encrypted_hex
  • encrypt() → ENCRYPTION_KEY yoksa HATA FIRLATIR (plain text yazmaz)
  • decrypt() → ENCRYPTION_KEY yoksa HATA FIRLATIR (sessizce geçmez)
  • Şifrelenen veriler: tüm AI platform API anahtarları, ApiKey tablosu

  KİMLİK DOĞRULAMA
  ─────────────────
  • NextAuth oturum tabanlı kimlik doğrulama
  • NEXTAUTH_SECRET ile JWT imzalanır
  • Şifre depolama: bcryptjs hash (plaintext asla saklanmaz)
  • Admin rotaları: getServerSession ile her istekte doğrulanır
  • SUPERADMIN rotaları: rol kontrolü (role !== 'SUPERADMIN' → 403)

  DİĞER ÖNLEMLER
  ───────────────
  • AUTOMATION_SECRET: otomasyon API'lerine dışarıdan erişim için token
  • PREVIEW_SECRET: taslak sayfa önizlemesi için token (boşsa 404 döner)
  • .env* → .gitignore'da (git'e asla eklenmez)
  • Yüklenen dosyalar /public/uploads/ → .gitignore'da
  • Cloudflare Worker URL/secret → .env üzerinden okunur (kod içinde sabit değer yok)
